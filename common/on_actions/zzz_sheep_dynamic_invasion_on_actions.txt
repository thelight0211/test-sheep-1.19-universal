on_actions = {
	on_monthly = {
		effect = {
			if = {
				limit = {
					is_major = yes
					is_ai = yes
					has_war = yes
					NOT = {
						any_enemy_country = {
							is_neighbor_of = ROOT
						}
					}
				}
				country_event = dynamic_invasion.1
			}
		}
	}
	on_war = {
		effect = {
			if = {
				limit = {
					is_major = yes
					is_ai = yes
					has_war = yes
					NOT = {
						any_enemy_country = {
							is_neighbor_of = ROOT
						}
					}
				}
				country_event = dynamic_invasion.1
			}
		}
	}
}
