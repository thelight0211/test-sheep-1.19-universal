on_actions = {
	on_monthly = {
		effect = {
			log = "[dynamic_invasion_debug_MONTHLY_TICK] country=[ROOT.GetTag]"
			if = {
				limit = {
					is_major = yes
					is_ai = yes
					has_war = yes
					NOT = {
						OR = {
							any_enemy_country = {
								is_neighbor_of = ROOT
							}
							any_allied_country = {
								any_enemy_country = {
									is_neighbor_of = PREV
								}
							}
						}
					}
					alliance_naval_strength_ratio < 1.5
				}
				country_event = dynamic_invasion.1
			}
		}
	}
	on_war = {
		effect = {
			if = {
				limit = {
					is_major = yes
					is_ai = yes
					NOT = {
						OR = {
							any_enemy_country = {
								is_neighbor_of = ROOT
							}
							any_allied_country = {
								any_enemy_country = {
									is_neighbor_of = PREV
								}
							}
						}
					}
					alliance_naval_strength_ratio < 1.5
				}
				country_event = dynamic_invasion.1
			}
		}
	}
}
