dynamic_distance_invasion_limits = {
	enable = {
		is_major = yes
		has_war = yes
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = invasion_unit_request
		state_trigger = {
			is_coastal = yes
			controller = {
				has_war_with = FROM
			}
			NOT = {
				is_in_array = {
					array = FROM.valid_naval_targets
					value = THIS
				}
			}
		}
		value = -250
	}

	ai_strategy = {
		type = invasion_unit_request
		state_trigger = {
			is_coastal = yes
			controller = {
				has_war_with = FROM
			}
			is_in_array = {
				array = FROM.valid_naval_targets
				value = THIS
			}
		}
		value = 1500
	}

	ai_strategy = {
		type = naval_invasion_focus
		value = 500
	}
}
