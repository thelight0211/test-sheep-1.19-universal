dynamic_distance_invasion_limits = {
	enable = {
		is_major = yes
		has_war = yes
		NOT = {
			OR = {
				any_enemy_country = {
					is_neighbor_of = ROOT
				}
				any_allied_country = {
					any_enemy_country = {
						is_neighbor_of = PREV
					}
				}
			}
		}
		alliance_naval_strength_ratio < 1.5
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = invasion_unit_request
		state_trigger = {
			is_coastal = yes
			owner = {
				has_war_with = FROM.FROM
			}
			NOT = {
				is_in_array = {
					array = FROM.FROM.valid_naval_targets
					value = THIS
				}
			}
			log = "[dynamic_invasion_debug] state=[THIS.GetName] owner=[THIS.owner.GetTag] FROM=[FROM.GetTag] FROM_FROM=[FROM.FROM.GetTag]"
		}
		value = -999999
	}
}
