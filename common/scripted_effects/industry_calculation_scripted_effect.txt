perform_the_industry_calculation_1 = {
	set_temp_variable = { ROOT.infantry_ratio = ROOT.num_battalions_with_type@infantry }
	set_temp_variable = { ROOT.tank_ratio = ROOT.num_battalions_with_type@light_armor }
	add_to_temp_variable = { ROOT.tank_ratio = ROOT.num_battalions_with_type@medium_armor }
	add_to_temp_variable = { ROOT.tank_ratio = ROOT.num_battalions_with_type@heavy_armor }
	add_to_temp_variable = { ROOT.tank_ratio = ROOT.num_battalions_with_type@light_tank_destroyer_brigade }
	add_to_temp_variable = { ROOT.tank_ratio = ROOT.num_battalions_with_type@medium_tank_destroyer_brigade }
	add_to_temp_variable = { ROOT.tank_ratio = ROOT.num_battalions_with_type@heavy_tank_destroyer_brigade }
	divide_temp_variable = { ROOT.tank_ratio = ROOT.infantry_ratio }
	if = {
		limit = {
			check_variable = {
				ROOT.tank_ratio < 0.20
			}
			NOT = { has_country_flag = unit_availability_flag }
		}	
		set_country_flag = unit_availability_flag
	}
	else_if = {
		limit = {
			check_variable = {
				ROOT.tank_ratio > 0.20
			}
			has_country_flag = unit_availability_flag
		}
		clr_country_flag = unit_availability_flag
	}
}

perform_the_industry_calculation_3 = {
	set_temp_variable = { ROOT.steel_max = ROOT.resource_produced@steel }
	set_temp_variable = { ROOT.steel_cons = ROOT.resource_consumed@steel }
	every_other_country = {
		limit = {
			OR = {
				is_in_faction_with = ROOT
				AND = {
					NOT = { has_war_with = ROOT }
					is_in_faction = no
				}
			}
			OR = {
				is_neighbor_of = ROOT
				any_controlled_state = {
					is_in_home_area = yes
					any_neighbor_state = {
						is_controlled_by = ROOT
						is_in_home_area = yes
					}
				}
				AND = {
					is_in_faction_with = ROOT
					ROOT = {
						has_equipment = { convoy > 200 }
					}
				}
			}
		}
		ROOT = {
			set_temp_variable = { PREV.steel_add = PREV.resource_produced@steel }
			divide_temp_variable = { PREV.steel_add = 2.5 }
			add_to_temp_variable = { ROOT.steel_max = PREV.steel_add }
		}
	}
	divide_temp_variable = { ROOT.steel_cons = ROOT.steel_max }
	if = {
		limit = {
			check_variable = {
				ROOT.steel_cons < 0.50
			}
			NOT = { has_country_flag = steel_availability_flag }
		}	
		set_country_flag = steel_availability_flag
	}
	else_if = {
		limit = {
			check_variable = {
				ROOT.steel_cons > 0.50
			}
			has_country_flag = steel_availability_flag
		}
		clr_country_flag = steel_availability_flag
	}
}
