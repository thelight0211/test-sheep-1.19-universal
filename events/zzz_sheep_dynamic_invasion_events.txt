add_namespace = dynamic_invasion

country_event = {
	id = dynamic_invasion.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		clear_array = valid_naval_targets
		clear_temp_array = root_coastal_ports

		every_state = {
			limit = {
				is_coastal = yes
				controller = {
					OR = {
						tag = ROOT
						is_ally_with = ROOT
					}
				}
			}
			add_to_temp_array = {
				temp_array = root_coastal_ports
				value = THIS
			}
		}

		set_temp_variable = {
			temp_var = found_target
			value = 0
		}

		every_state = {
			limit = {
				is_coastal = yes
				controller = {
					has_war_with = ROOT
				}
				ROOT = {
					any_of_scopes = {
						array = root_coastal_ports
						distance_to = {
							target = PREV.PREV
							value < 200
						}
					}
				}
			}
			ROOT = {
				add_to_array = {
					array = valid_naval_targets
					value = PREV
				}
				set_temp_variable = {
					temp_var = found_target
					value = 1
				}
			}
		}

		if = {
			limit = {
				check_variable = {
					temp_var = found_target
					value = 0
				}
			}
			every_state = {
				limit = {
					is_coastal = yes
					controller = {
						has_war_with = ROOT
					}
					ROOT = {
						any_of_scopes = {
							array = root_coastal_ports
							distance_to = {
								target = PREV.PREV
								value < 450
							}
						}
					}
				}
				ROOT = {
					add_to_array = {
						array = valid_naval_targets
						value = PREV
					}
					set_temp_variable = {
						temp_var = found_target
						value = 1
					}
				}
			}
		}

		if = {
			limit = {
				check_variable = {
					temp_var = found_target
					value = 0
				}
			}
			every_state = {
				limit = {
					is_coastal = yes
					controller = {
						has_war_with = ROOT
					}
					ROOT = {
						any_of_scopes = {
							array = root_coastal_ports
							distance_to = {
								target = PREV.PREV
								value < 1000
							}
						}
					}
				}
				ROOT = {
					add_to_array = {
						array = valid_naval_targets
						value = PREV
					}
					set_temp_variable = {
						temp_var = found_target
						value = 1
					}
				}
			}
		}

		clear_temp_array = root_coastal_ports
	}
}
