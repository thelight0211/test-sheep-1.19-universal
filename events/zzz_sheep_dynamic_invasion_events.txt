add_namespace = dynamic_invasion

country_event = {
	id = dynamic_invasion.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		clear_array = valid_naval_targets

		every_state = {
			limit = {
				is_coastal = yes
				controller = {
					has_war_with = ROOT
				}
			}
			save_event_target_as = candidate_target
			if = {
				limit = {
					OR = {
						# ROOT's own controlled coastal states within 1200
						ROOT = {
							any_controlled_state = {
								is_coastal = yes
								distance_to = {
									target = event_target:candidate_target
									value < 1200
								}
							}
						}
						# Allied coastal states within 1200
						any_country = {
							is_ally_with = ROOT
							has_capitulated = no
							any_controlled_state = {
								is_coastal = yes
								distance_to = {
									target = event_target:candidate_target
									value < 1200
								}
							}
						}
					}
				}
				ROOT = {
					add_to_array = {
						array = valid_naval_targets
						value = event_target:candidate_target
					}
				}
			}
		}

		# Debug log — check game.log to verify array population.
		# Remove this line after testing.
		log = "SHEEP_INVASION [Root.GetName]: valid_naval_targets populated with [?valid_naval_targets.GetCount] states"
	}
}
