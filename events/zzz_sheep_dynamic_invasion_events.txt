add_namespace = dynamic_invasion

country_event = {
	id = dynamic_invasion.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		clear_array = valid_naval_targets
		clear_temp_array = root_coastal_ports

		every_state = {
			limit = {
				is_coastal = yes
				controller = {
					OR = {
						tag = ROOT
						AND = {
							is_ally_with = ROOT
							has_capitulated = no
						}
					}
				}
			}
			add_to_temp_array = {
				temp_array = root_coastal_ports
				value = THIS
			}
		}

		every_state = {
			limit = {
				is_coastal = yes
				controller = {
					has_war_with = ROOT
				}
			}
			save_event_target_as = target_enemy_state
			if = {
				limit = {
					ROOT = {
						any_of_scopes = {
							array = root_coastal_ports
							distance_to = {
								target = event_target:target_enemy_state
								value < 1200
							}
						}
					}
				}
				ROOT = {
					add_to_array = {
						array = valid_naval_targets
						value = event_target:target_enemy_state
					}
				}
			}
		}

		clear_temp_array = root_coastal_ports
	}
}
